# Two rule groups: classic search crawlers, then AI retrieval/agent crawlers.
# A crawler obeys only its MOST SPECIFIC matching group, so every group repeats
# the full rule set instead of inheriting from `*`.

# NOT disallowed on purpose: /checkout, /confirm, /review/ and /search.
# Every one of them already serves `noindex` (meta + X-Robots-Tag, in BOTH
# locales — verified). Blocking them here would stop Googlebot fetching the
# page and therefore stop it ever READING that noindex, which is how a URL ends
# up in the index with no snippet — Search Console reported exactly that for
# /en/checkout?add_sku=JR-TC30W under "Indexed, though blocked by robots.txt".
# It also removes a locale asymmetry: `Disallow: /checkout` never matched
# `/en/checkout`, so the English funnel was crawlable while the Arabic one was
# not. noindex is the correct tool for de-indexing; robots.txt is not.
Disallow: /admin/
Disallow: /api/admin/
Disallow: /api/categories
Disallow: /api/indexing
Disallow: /api/orders
Disallow: /api/reviews
Disallow: /api/seed
Disallow: /api/serials
Disallow: /api/v1/
# Markdown twin of every HTML page (Accept: text/markdown negotiation target).
# Search engines already receive the HTML original at the canonical URL, so
# crawling the alternate representation is pure crawl-budget waste. AI agents
# keep explicit access in the group below.
Disallow: /api/markdown-negotiate

# Allow lines for /api/v1/checkout and /api/v1/quick-cod override /api/v1/ Disallow (Google longest-match).
